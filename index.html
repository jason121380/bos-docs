function findUniqueTargetTicket(originTicket, targetTickets, oldStoreName) {
  const originNo = normalizeTicketTransferText(originTicket.no);
  const originName = normalizeTicketTransferText(originTicket.name);
  const prefixedNo = originNo ? `[${oldStoreName}]${originNo}` : '';
  const prefixedName = `[${oldStoreName}]${originName}`;
  const matchers = [];

  if (originNo) {
    matchers.push({
      name: `代號 ${originNo}`,
      match: targetTicket => normalizeTicketTransferText(targetTicket.no) === originNo
    });
    matchers.push({
      name: `轉移代號 ${prefixedNo}`,
      match: targetTicket => normalizeTicketTransferText(targetTicket.no) === prefixedNo
    });
  }

  matchers.push({
    name: `名稱/價格/次數 ${originName}`,
    match: targetTicket => ticketTransferSignature(targetTicket) === ticketTransferSignature(originTicket)
  });
  matchers.push({
    name: `轉移名稱/價格/次數 ${prefixedName}`,
    match: targetTicket => ticketTransferSignature(targetTicket) === ticketTransferSignature(originTicket, prefixedName)
  });

  for (const matcher of matchers) {
    const matchedTickets = targetTickets.filter(matcher.match);
    if (matchedTickets.length > 1) {
      throw `新店卡券設定「${originTicket.name}」以${matcher.name}比對時找到多筆，請先整理重複卡券設定`;
    }
    if (matchedTickets.length === 1) return matchedTickets[0];
  }

  return null;
}

/**
 * 將原始 CSV 檔案過濾並輸出符合 active:inactive = 4:6 的比例
 * @param {string} inputFilePath - 原始 CSV 檔案路徑
 * @param {string} outputFilePath - 輸出 CSV 檔案路徑
 */
function filterPubchemCsv(inputFilePath, outputFilePath) {
  const activeRows = [];
  const inactiveRows = [];

  fs.createReadStream(inputFilePath)
    .pipe(csv())
    .on('data', (row) => {
      const outcome = row['PUBCHEM_ACTIVITY_OUTCOME'];
      if (outcome === 'Active') {
        activeRows.push(row);
      } else if (outcome === 'Inactive') {
        inactiveRows.push(row);
      }
    })
    .on('end', () => {
      const inactiveSampleSize = Math.floor(activeRows.length * 1.5);
      const sampledInactive = shuffle(inactiveRows).slice(0, inactiveSampleSize);
      const combined = shuffle([...activeRows, ...sampledInactive]);

      // 自動產生欄位定義
      const headers = Object.keys(combined[0]).map((key) => ({ id: key, title: key }));
      const csvWriter = createObjectCsvWriter({
        path: outputFilePath,
        header: headers,
      });

      csvWriter
        .writeRecords(combined)
        .then(() => {
          console.log(`✅ CSV 檔已產出至：${outputFilePath}`);
        })
        .catch((err) => console.error('寫入 CSV 時發生錯誤：', err));
    });
}

// 工具函式：隨機打亂陣列（Fisher-Yates Shuffle）
function shuffle(array) {
  const result = [...array];
  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]];
  }
  return result;
}

function fullwidthToHalfwidth(str) {
  // 建立全形數字和半形數字的映射
  const fullwidthChars = '０１２３４５６７８９'
  const halfwidthChars = '0123456789'
  const charMap = {}
  for (let i = 0; i < fullwidthChars.length; i++) {
    charMap[fullwidthChars[i]] = halfwidthChars[i]
  }
  // 使用映射表將全形數字替換為半形數字
  let result = ''
  for (let i = 0; i < str.length; i++) {
    const char = str[i]
    result += charMap[char] || char
  }
  return result
}

 * @param {Object} params
 * @param {Array<string>} params.targetStoreIds - 目標店家 ID 陣列
 * @param {Object} params.base - 基礎設定
 * @param {string} params.base.storeId - 來源店家 ID
 * @param {string} params.base.productCategoryId - 來源商品類別 ID
 * @param {string} params.creatorId - 建立者 ID
 * @returns {Promise<{success: boolean, message: string, details: Object}>}
 */
async function copyProductCategoryAndShortcuts({ targetStoreIds, base, creatorId }) {
  
  const t = await db.sequelize.transaction()
  
  try {
    // 1. 檢索來源商品類別
    const baseCategory = await db.ProductCategory.findOne({
      where: {
        OfficialBrandMerchantId: base.storeId,
        id: base.productCategoryId
      },
      transaction: t
    })
    
    if (!baseCategory) {
      throw new Error(`找不到來源商品類別 (storeId: ${base.storeId}, categoryId: ${base.productCategoryId})`)
    }
    
    console.log(`找到來源商品類別: ${baseCategory.name} (${baseCategory.no})`)
    
    // 2. 檢索來源商品
    const baseProducts = await db.Product.findAll({
      where: {
        OfficialBrandMerchantId: base.storeId,
        ProductCategoryId: base.productCategoryId
      },
      transaction: t
    })
    
    console.log(`找到 ${baseProducts.length} 個來源商品`)
    
    // 3. 檢索來源組合設定（包含 ShortcutProducts）
    const baseShortcuts = await db.Shortcut.findAll({
      where: {
        OfficialBrandMerchantId: base.storeId
      },
      include: [
        { model: db.ShortcutProduct }
      ],
      transaction: t
    })
    
    console.log(`找到 ${baseShortcuts.length} 個來源組合設定`)
    
    // 建立來源商品 ID 到商品資訊的對應 Map
    const baseProductMap = new Map()
    baseProducts.forEach(product => {
      baseProductMap.set(product.id, product)
    })
    
    // 記錄處理結果
    const results = {
      categories: [],
      products: [],
      shortcuts: []
    }
    
    // 4. 為每個目標店家建立資料
    for (const targetStoreId of targetStoreIds) {
      console.log(`\n開始處理目標店家: ${targetStoreId}`)
      
      // 4.1 建立商品類別
      let targetCategory = await db.ProductCategory.findOne({
        where: {
          OfficialBrandMerchantId: targetStoreId,
          name: baseCategory.name,
          no: baseCategory.no
        },
        transaction: t
      })
      
      if (!targetCategory) {
        targetCategory = await db.ProductCategory.create({
          no: baseCategory.no,
          name: baseCategory.name,
          isUsing: baseCategory.isUsing,
          OfficialBrandMerchantId: targetStoreId,
          CreatorId: creatorId
        }, { transaction: t })
        
        console.log(`建立商品類別: ${targetCategory.name}`)
        results.categories.push({
          storeId: targetStoreId,
          categoryId: targetCategory.id,
          name: targetCategory.name,
          action: 'created'
        })
      } else {
        console.log(`商品類別已存在: ${targetCategory.name}`)
        results.categories.push({
          storeId: targetStoreId,
          categoryId: targetCategory.id,
          name: targetCategory.name,
          action: 'existed'
        })
      }
      
      // 4.2 建立商品（並建立舊商品ID到新商品ID的對應關係）
      const productIdMapping = new Map() // 來源商品ID -> 目標商品ID
      
      for (const baseProduct of baseProducts) {
        // 檢查商品是否已存在
        let targetProduct = await db.Product.findOne({
          where: {
            OfficialBrandMerchantId: targetStoreId,
            ProductCategoryId: targetCategory.id,
            no: baseProduct.no,
            name: baseProduct.name
          },
          transaction: t
        })
        
        if (!targetProduct) {
          targetProduct = await db.Product.create({
            no: baseProduct.no,
            name: baseProduct.name,
            cost: baseProduct.cost,
            sellPrice: baseProduct.sellPrice,
            employeePrice: baseProduct.employeePrice,
            stockCount: baseProduct.stockCount,
            stockUnit: baseProduct.stockUnit,
            safeHigh: baseProduct.safeHigh,
            safeLow: baseProduct.safeLow,
            commissionWay: baseProduct.commissionWay,
            commission: baseProduct.commission,
            customerSellSetting: baseProduct.customerSellSetting,
            isUsing: baseProduct.isUsing,
            isAvailableForSale: baseProduct.isAvailableForSale,
            isLocked: baseProduct.isLocked,
            isPublished: baseProduct.isPublished,
            ProductCategoryId: targetCategory.id,
            OfficialBrandMerchantId: targetStoreId,
            CreatorId: creatorId,
            VendorId: baseProduct.VendorId
          }, { transaction: t })
          
          console.log(`建立商品: ${targetProduct.name}`)
          results.products.push({
            storeId: targetStoreId,
            productId: targetProduct.id,
            name: targetProduct.name,
            action: 'created'
          })
        } else {
          console.log(`商品已存在: ${targetProduct.name}`)
          results.products.push({
            storeId: targetStoreId,
            productId: targetProduct.id,
            name: targetProduct.name,
            action: 'existed'
          })
        }
        
        // 記錄來源商品ID到目標商品ID的對應
        productIdMapping.set(baseProduct.id, targetProduct.id)
      }
      
      // 4.3 建立組合設定
      for (const baseShortcut of baseShortcuts) {
        // 過濾出該組合中屬於指定商品類別的商品
        const relatedShortcutProducts = baseShortcut.ShortcutProducts.filter(sp => 
          baseProductMap.has(sp.ProductId)
        )
        
        // 如果該組合沒有包含指定類別的商品，則跳過
        if (relatedShortcutProducts.length === 0) {
          continue
        }
        
        // 檢查組合是否已存在
        let targetShortcut = await db.Shortcut.findOne({
          where: {
            OfficialBrandMerchantId: targetStoreId,
            name: baseShortcut.name,
            no: baseShortcut.no
          },
          include: [{ model: db.ShortcutProduct }],
          transaction: t
        })
        
        if (!targetShortcut) {
          // 建立新組合
          const newShortcutProducts = relatedShortcutProducts
            .filter(sp => productIdMapping.has(sp.ProductId))
            .map(sp => ({
              ProductId: productIdMapping.get(sp.ProductId)
            }))
          
          targetShortcut = await db.Shortcut.create({
            no: baseShortcut.no,
            name: baseShortcut.name,
            OfficialBrandMerchantId: targetStoreId,
            CreatorId: creatorId,
            ShortcutProducts: newShortcutProducts
          }, {
            transaction: t,
            include: [{ model: db.ShortcutProduct }]
          })
          
          console.log(`建立組合: ${targetShortcut.name}`)
          results.shortcuts.push({
            storeId: targetStoreId,
            shortcutId: targetShortcut.uuid,
            name: targetShortcut.name,
            productCount: newShortcutProducts.length,
            action: 'created'
          })
        } else {
          // 組合已存在，為其添加缺少的商品
          const existingProductIds = new Set(
            targetShortcut.ShortcutProducts.map(sp => sp.ProductId)
          )
          
          let addedCount = 0
          for (const sp of relatedShortcutProducts) {
            const targetProductId = productIdMapping.get(sp.ProductId)
            if (targetProductId && !existingProductIds.has(targetProductId)) {
              await db.ShortcutProduct.create({
                ShortcutId: targetShortcut.id,
                ProductId: targetProductId
              }, { transaction: t })
              addedCount++
            }
          }
          
          console.log(`組合已存在: ${targetShortcut.name}，新增 ${addedCount} 個商品`)
          results.shortcuts.push({
            storeId: targetStoreId,
            shortcutId: targetShortcut.uuid,
            name: targetShortcut.name,
            addedProductCount: addedCount,
            action: 'updated'
          })
        }
      }
    }
    
    // 5. 提交交易
    await t.commit()
    
    console.log('\n=== 複製完成 ===')
    console.log(`商品類別: ${results.categories.length} 筆`)
    console.log(`商品: ${results.products.length} 筆`)
    console.log(`組合: ${results.shortcuts.length} 筆`)
    
    return {
      success: true,
      message: '複製完成',
      details: results
    }
    
  } catch (err) {
    console.log('copyProductCategoryAndShortcuts 錯誤:', err)
    await t.rollback()
    return {
      success: false,
      message: `複製失敗: ${err.message}`,
      details: null
    }
  }
}

function storeSort(arr) {
  return arr.sort((a, b) => {
    const targetKeyA = a.storeName
    const targetKeyB = b.storeName
    // 檢查第一個字符是否為英文字母
    const isAEnglish = /^[a-zA-Z]/.test(targetKeyA);
    const isBEnglish = /^[a-zA-Z]/.test(targetKeyB);
    // 如果其中一個是英文開頭，另一個不是，英文開頭的排在前面
    if (isAEnglish && !isBEnglish) return -1;
    if (!isAEnglish && isBEnglish) return 1;
    // 如果兩者都是英文開頭或都不是英文開頭，則按字典順序排序
    return targetKeyA.localeCompare(targetKeyB, undefined, {
      numeric: true,          // 數字排序
      sensitivity: 'base',    // 不區分大小寫和重音符號
      ignorePunctuation: true // 忽略標點符號
    });
  });

function storeSort(arr) {
  return arr.sort((a, b) => {
    // 檢查第一個字符是否為英文字母
    const isAEnglish = /^[a-zA-Z]/.test(a);
    const isBEnglish = /^[a-zA-Z]/.test(b);
    // 如果其中一個是英文開頭，另一個不是，英文開頭的排在前面
    if (isAEnglish && !isBEnglish) return -1;
    if (!isAEnglish && isBEnglish) return 1;
    // 如果兩者都是英文開頭或都不是英文開頭，則按字典順序排序
    return a.localeCompare(b, undefined, {
      numeric: true,          // 數字排序
      sensitivity: 'base',    // 不區分大小寫和重音符號
      ignorePunctuation: true // 忽略標點符號
    });
  });
}
